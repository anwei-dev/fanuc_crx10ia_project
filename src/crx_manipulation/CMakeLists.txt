cmake_minimum_required(VERSION 3.16)
project(crx_manipulation)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo REQUIRED)

add_library(crx_cube_grasp SHARED src/cube_grasp_plugin.cpp)
target_include_directories(crx_cube_grasp SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(crx_cube_grasp ${GAZEBO_LIBRARIES})
ament_target_dependencies(crx_cube_grasp gazebo_ros rclcpp std_srvs)

add_executable(fixed_pick_place src/fixed_pick_place.cpp)
ament_target_dependencies(fixed_pick_place rclcpp rclcpp_action control_msgs trajectory_msgs
  geometry_msgs shape_msgs moveit_msgs moveit_core moveit_ros_planning_interface gazebo_msgs std_srvs)

install(TARGETS crx_cube_grasp LIBRARY DESTINATION lib)
install(TARGETS fixed_pick_place DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config worlds DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(scene_geometry test/test_scene_geometry.py)
endif()
ament_package()
